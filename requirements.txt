# SVG -> PNG rendering (scripts/make_cards.py, scripts/build_diagrams.py).
# Imported lazily in make_cards.py, so the CI `--check-only` guard does not
# need it.
#
# These two are PINNED while the rest of this file is not, because they are
# the only deps whose output is committed to the repo as bytes. The generated
# PNGs under assets/cards/, assets/if/, assets/oia/ and assets/sda/ are
# checked in, and any change in the rasterizer -- font hinting, antialiasing,
# gamma -- rewrites every one of them even when no SVG changed. Unpinned, that
# drift lands in whatever unrelated commit happens to run a regeneration.
# Moving these pins is therefore a deliberate act that gets its own re-render
# commit.

cairocffi==1.7.1
